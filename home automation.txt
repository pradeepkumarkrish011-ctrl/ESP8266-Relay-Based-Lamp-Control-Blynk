#define BLYNK_TEMPLATE_NAME "Home Automation"
#define BLYNK_AUTH_TOKEN "QkAAzaYnH_9ynFdxIOmvvNf0p3S1PS8o"
#define BLYNK_PRINT Serial

char ssid[] = "**********";           
char pass[] = "**********";   

#define RELAY1_PIN  D1   // GPIO5

BLYNK_WRITE(V0)
{
  int value = param.asInt();
  if (value == 1) 
  {
    digitalWrite(RELAY1_PIN, LOW);   // Relay ON
  } else 
  {
    digitalWrite(RELAY1_PIN, HIGH);  // Relay OFF
  }
}

void setup() 
{
  Serial.begin(115200);
  pinMode(RELAY1_PIN, OUTPUT);
  digitalWrite(RELAY1_PIN, HIGH); // Relay OFF initially
  Blynk.begin(BLYNK_AUTH_TOKEN, ssid, pass);
}

void loop() 
{
  Blynk.run();
}
